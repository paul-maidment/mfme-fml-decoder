Which lamps belong to which borders?

Worked report using the exact MFME component-list examples. Revised conclusion: with X, Y, W, H, sibling local offsets, and z-order when copies compete, every scenario below resolves. Do not require lamp.z < border.z.

Revised decision rule

  1. Containment. A lamp is a candidate for a border only if the lamp rectangle sits inside the border rectangle. If it sits in none, it is an orphan (belongs to no border).
  2. Unique owner. If only one border contains the lamp, assign it there.
  3. Local offset template. Same-sized sibling borders own the same set of local offsets (dx, dy) = (lamp.X − border.X, lamp.Y − border.Y). In these examples the template is (+1, +1) and (+21, +57).
  4. Z-order for competing copies. Process borders by rising z. For each template offset, assign the lowest-z unmatched lamp that is contained in that border and matches the offset. Do not require the lamp to be below the border in z — a lamp painted on top of its border is still allowed.

Plain language: stickers go on frames they’re inside; matching stickers sit in the same place inside each copy of the frame; when copies are stacked, draw order picks which sticker goes with which frame. Sticker above the frame is fine.

“Offsets required” means containment / unique screen position alone is ambiguous (e.g. two lamps on one pixel); you must use (dx, dy) relative to each border’s origin.

C# reference implementation

Same revised rule, expressed in C#. Wired into the decoder via BorderOwnershipAssigner after all components are parsed; each border emits OwnedOrdinalComponentIdentifiers. A “child” may be any non-border component (lamp, button, etc.). Template offsets are discovered from sibling ownership (fallback (+1, +1) / (+21, +57) for the 39×75 examples). 

using System;
using System.Collections.Generic;
using System.Linq;

public readonly record struct Rect(int X, int Y, int Width, int Height)
{
    public bool Contains(Rect other) =>
        other.X >= X
        && other.Y >= Y
        && other.X + other.Width <= X + Width
        && other.Y + other.Height <= Y + Height;
}

public readonly record struct Offset(int Dx, int Dy);

public sealed class LayoutComponent
{
    public string Id { get; init; }
    public string Kind { get; init; }  // e.g. "Border", "Lamp", or any other type
    public Rect Bounds { get; init; }
    public int OrdinalComponentIdentifier { get; init; }
}

public sealed class BorderOwnershipResult
{
    public IReadOnlyDictionary<LayoutComponent, List<LayoutComponent>> Assignment { get; init; }
    public IReadOnlyList<LayoutComponent> Orphans { get; init; }
    public IReadOnlyList<LayoutComponent> Unresolved { get; init; }
}

/// <summary>
/// Assigns candidate components to border containers by containment,
/// sibling local-offset templates, and z-order when copies compete.
/// Candidates may be lamps, other borders, or any layout component.
/// </summary>
public static class BorderOwnershipAssigner
{
    // Known sibling template for the 39×75 border examples in this report.
    private static readonly Offset[] DefaultTemplate =
    {
        new Offset(1, 1),
        new Offset(21, 57),
    };

    public static BorderOwnershipResult Assign(
        IReadOnlyList<LayoutComponent> borders,
        IReadOnlyList<LayoutComponent> candidates,
        IReadOnlyList<Offset> template = null)
    {
        if (borders is null) throw new ArgumentNullException(nameof(borders));
        if (candidates is null) throw new ArgumentNullException(nameof(candidates));

        Offset[] slots = (template ?? DefaultTemplate).ToArray();
        var assignment = borders.ToDictionary(b => b, _ => new List<LayoutComponent>());
        var unmatched = new HashSet<LayoutComponent>(candidates);
        var orphans = new List<LayoutComponent>();

        // 1. Orphans — contained by no border
        foreach (LayoutComponent component in candidates)
        {
            if (!borders.Any(b => b.Bounds.Contains(component.Bounds)))
            {
                orphans.Add(component);
                unmatched.Remove(component);
            }
        }

        // 2. Unique containment
        foreach (LayoutComponent component in unmatched.ToList())
        {
            List<LayoutComponent> owners = borders
                .Where(b => b.Bounds.Contains(component.Bounds))
                .ToList();

            if (owners.Count == 1)
            {
                assignment[owners[0]].Add(component);
                unmatched.Remove(component);
            }
        }

        // 3. Competing copies — fill template slots by rising border z
        //    Do NOT require component.OrdinalComponentIdentifier < border.OrdinalComponentIdentifier
        foreach (LayoutComponent border in borders.OrderBy(b => b.OrdinalComponentIdentifier))
        {
            foreach (Offset slot in slots)
            {
                LayoutComponent match = unmatched
                    .Where(c => border.Bounds.Contains(c.Bounds)
                                && LocalOffset(border, c).Equals(slot))
                    .OrderBy(c => c.OrdinalComponentIdentifier)
                    .FirstOrDefault();

                if (match is null)
                    continue;

                assignment[border].Add(match);
                unmatched.Remove(match);
            }
        }

        return new BorderOwnershipResult
        {
            Assignment = assignment,
            Orphans = orphans,
            Unresolved = unmatched.OrderBy(c => c.OrdinalComponentIdentifier).ToList(),
        };
    }

    public static Offset LocalOffset(LayoutComponent border, LayoutComponent child) =>
        new Offset(child.Bounds.X - border.Bounds.X, child.Bounds.Y - border.Bounds.Y);
}

// --- Scenario 5 worked example ---
//
// Border z=5 → components z=1 (+1,+1), z=3 (+21,+57)
// Border z=6 → components z=7 (+1,+1), z=4 (+21,+57)
// Component z=2 at (380,182) → orphan

public static class Scenario5Example
{
    public static BorderOwnershipResult Run()
    {
        var borders = new List<LayoutComponent>
        {
            Comp("B5", "Border", 224, 113, 39, 75, z: 5),
            Comp("B6", "Border", 224, 113, 39, 75, z: 6),
        };

        // Any component kinds are valid candidates — lamps used here only as the worked example.
        var candidates = new List<LayoutComponent>
        {
            Comp("C1", "Lamp", 225, 114, 17, 17, z: 1),
            Comp("C2", "Lamp", 380, 182, 17, 17, z: 2), // outside
            Comp("C3", "Lamp", 245, 170, 17, 17, z: 3),
            Comp("C4", "Lamp", 245, 170, 17, 17, z: 4),
            Comp("C7", "Lamp", 225, 114, 17, 17, z: 7), // above borders
        };

        return BorderOwnershipAssigner.Assign(borders, candidates);
    }

    private static LayoutComponent Comp(
        string id, string kind, int x, int y, int w, int h, int z) =>
        new LayoutComponent
        {
            Id = id,
            Kind = kind,
            Bounds = new Rect(x, y, w, h),
            OrdinalComponentIdentifier = z,
        };
}

Bottom line: for every scenario provided so far, the revised rule resolves which components belong to which border whenever z-order is present for stacked clones. Geometry + offsets handle the rest. Outside lamps correctly stay unassigned.
